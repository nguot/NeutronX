Phụ lục này mô tả một số use case không được trình bày chi tiết trong phần thân, gồm bước cấp quyền chi tiêu một lần, bước kiểm tra đăng ký của filler, và hai API truy vấn giá của backend.

\section{Cấp quyền chi tiêu một lần qua Permit2}

\begin{itemize}
    \item \textbf{Tác nhân}: Người dùng (swapper).
    \item \textbf{Tiền điều kiện}: Người dùng có token ERC-20 muốn bán và đã kết nối ví.
    \item \textbf{Luồng chính}:
    \begin{enumerate}
        \item Người dùng gọi \texttt{approve} cho hợp đồng Permit2 trên token muốn bán (làm một lần cho mỗi token).
        \item Người dùng cấp hạn mức cho hợp đồng cần dùng (Reactor hoặc EscrowSrcFactory) thông qua Permit2.
    \end{enumerate}
    \item \textbf{Hậu điều kiện}: Hợp đồng được phép kéo token của người dùng tới đúng địa chỉ đích khi lệnh được khớp, mà người dùng không phải ký lại hay trả phí gas cho từng lần khớp.
\end{itemize}

Đây là lý do người dùng ``khóa'' tài sản mà không cần gửi trước vào một hợp đồng trung gian: tài sản vẫn nằm trong ví cho tới đúng lúc một phần lệnh được khớp.

\begin{example}
Người dùng có 5\,000 USDC và muốn đổi dần thành WETH. Họ gọi \texttt{usdc.approve(permit2, type(uint256).max)} một lần duy nhất. Sau đó, mỗi lần tạo lệnh 1\,000 USDC, họ cấp hạn mức đúng 1\,000 USDC cho Reactor qua Permit2. Phần còn lại (4\,000 USDC) vẫn nằm trong ví, không bị khóa, và được rút khi từng lệnh tiếp theo được tạo và khớp thành công.
\end{example}

\section{Kiểm tra đăng ký hợp lệ của filler}

\begin{itemize}
    \item \textbf{Tác nhân}: Filler, hợp đồng \texttt{PartialFillReactor}.
    \item \textbf{Tiền điều kiện}: Filler đã đăng ký và đặt cọc cho một phần lệnh qua \texttt{FillAuction}.
    \item \textbf{Luồng chính}:
    \begin{enumerate}
        \item Filler gọi hàm khớp lệnh với khối lượng mong muốn.
        \item Hợp đồng kiểm tra qua \texttt{hasValidRegistration}: đăng ký đúng là của filler này, chưa tất toán, chưa bị phạt, khối lượng xin khớp không vượt phần đã cam kết, và chưa quá hạn.
        \item Nếu hợp lệ, hợp đồng tiếp tục khớp; nếu không, giao dịch bị từ chối.
    \end{enumerate}
    \item \textbf{Hậu điều kiện}: Duy nhất filler có đăng ký hợp lệ được phép khớp lệnh; các bên không đặt cọc không thể tham gia.
\end{itemize}

\begin{example}
Có một lệnh tổng 10\,000 USDC. Filler A đăng ký cam kết khớp 60\% (6\,000 USDC) và đặt cọc tương ứng. Khi gọi \texttt{executePartialChunk} với \texttt{fillAmount = 6000 USDC}, hợp đồng xác nhận đúng cam kết và cho phép. Nếu Filler A sau đó thử gọi thêm \texttt{fillAmount = 500 USDC} (vượt phần đã đăng ký), giao dịch bị từ chối. Filler B có thể đăng ký phần còn lại (4\,000 USDC) một cách độc lập.
\end{example}

\section{Truy vấn giá thị trường}

\begin{itemize}
    \item \textbf{Tác nhân}: Frontend hoặc filler muốn xem giá trước khi tạo lệnh.
    \item \textbf{Tiền điều kiện}: Backend đang chạy và có thể kết nối đến ít nhất một aggregator (Uniswap, KyberSwap...).
    \item \textbf{Tham số}: \texttt{inputToken}, \texttt{outputToken}, \texttt{inputAmount}, \texttt{inputDecimals}, \texttt{outputDecimals}.
    \item \textbf{Luồng chính}:
    \begin{enumerate}
        \item Frontend gửi \texttt{GET /quote} với thông tin cặp token và khối lượng.
        \item Backend lấy giá từ aggregator tốt nhất hiện có, tính giá thị trường theo TWAP và ước tính số token nhận được.
        \item Backend trả về \texttt{estimatedOutput}, \texttt{marketRate}, \texttt{priceImpact} và nguồn aggregator đã dùng.
    \end{enumerate}
    \item \textbf{Hậu điều kiện}: Người dùng thấy được ước tính số token nhận trước khi quyết định tạo lệnh.
\end{itemize}

\begin{example}
Người dùng muốn biết đổi 1 WETH được bao nhiêu USDC. Frontend gọi \texttt{GET /quote} với tham số:
\begin{lstlisting}[numbers=none]
{
  "inputToken":  "WETH",
  "outputToken": "USDC",
  "inputAmount": "1000000000000000000"
}
\end{lstlisting}
Backend truy vấn Uniswap và trả về:
\begin{lstlisting}[numbers=none]
{
  "estimatedOutput": "3850000000",
  "marketRate":      "3850.00",
  "priceImpact":     "0.12%",
  "source":          "uniswap"
}
\end{lstlisting}
Người dùng thấy ngay giá ước tính (3\,850 USDC) và mức trượt giá trước khi xác nhận.
\end{example}

\section{Gợi ý tham số lệnh tự động}

\begin{itemize}
    \item \textbf{Tác nhân}: Frontend khi người dùng muốn tạo lệnh mà không tự chỉnh tham số đường cong giá.
    \item \textbf{Tiền điều kiện}: Có ít nhất một filler đăng ký phục vụ cặp token tương ứng.
    \item \textbf{Tham số}: \texttt{inputToken}, \texttt{outputToken}, \texttt{inputAmount}; tuỳ chọn: \texttt{deadlineBlocks} (mặc định 100), \texttt{targetCoverageBps} (mặc định 9000, tức 90\%), \texttt{premiumBps} (mặc định 200, tức +2\%), \texttt{slippageBps} (mặc định 100, tức --1\%).
    \item \textbf{Luồng chính}:
    \begin{enumerate}
        \item Backend lấy giá thị trường hiện tại và số block hiện tại.
        \item Backend tính \texttt{startPrice} (giá khởi điểm cao hơn thị trường \texttt{premiumBps}), \texttt{minOutputAmount} (sàn tuyệt đối thấp hơn thị trường \texttt{slippageBps}), và \texttt{decayPerBlock} sao cho giá giảm đều từ đỉnh xuống sàn trong \texttt{deadlineBlocks} block.
        \item Backend tìm kiếm nhị phân trên danh sách \texttt{minFillBps} để chọn giá trị lớn nhất (ít phân mảnh nhất) mà các filler đang đăng ký vẫn đáp ứng được ít nhất \texttt{targetCoverageBps}\% khối lượng.
        \item Backend trả về toàn bộ tham số sẵn sàng để tạo lệnh: \texttt{startPrice}, \texttt{decayPerBlock}, \texttt{minOutputAmount}, \texttt{deadline}, \texttt{minFillBps}, kèm ghi chú về mức độ phủ của các filler.
    \end{enumerate}
    \item \textbf{Hậu điều kiện}: Người dùng nhận được bộ tham số tối ưu có thể dùng trực tiếp để tạo lệnh mà không cần hiểu chi tiết về đường cong đấu giá Hà Lan.
\end{itemize}

\begin{example}
Người dùng muốn đổi 1 WETH, thị trường đang ở 3\,850 USDC/WETH. Backend tự động tính:
\begin{itemize}
    \item \texttt{startPrice = 3\,850 × 1{,}02 = 3\,927 USDC} (cao hơn thị trường 2\% để hấp dẫn filler nhanh).
    \item \texttt{minOutputAmount = 3\,850 × 0{,}99 = 3\,811 USDC} (sàn tuyệt đối thấp hơn thị trường 1\%).
    \item \texttt{decayPerBlock = (3927 - 3811) / 100 = 1,16 USDC/block} (giá giảm đều trong 100 block).
    \item \texttt{minFillBps = 2000} (20\%) — giá trị lớn nhất mà các filler đang đăng ký vẫn có thể phủ ít nhất 90\% khối lượng.
\end{itemize}
Người dùng nhận bộ tham số hoàn chỉnh và xác nhận ký lệnh.
\end{example}
